public class TwoChecksPredicate {

	public static void main(String[] args) {
		Integer[] arr= {12,3,4,5,6,22,19,10,14,33,67,15};
		
		Predicate<Integer> p1= i -> i%2==0;
		Predicate<Integer> p2= i -> i>10;
		
		System.out.println("numbers are even and greater than 10");
		for(int i:arr) {
			if(p1.and(p2).test(i)) {
				System.out.println(i);
			}
		}
		
		System.out.println("numbers are even or  greater than 10");
		
		
		for(int i:arr) {
			if(p1.or(p2).test(i)) {
				System.out.println(i);
			}
		}
		
		
		System.out.println("Not even numbers");
		for(int i:arr) {
			if(p1.negate().test(i)) {
				System.out.println(i);
			}
		}
	}

}

public class Student {
	int marks;
	String name;
	public Student(int marks, String name) {
		
		this.marks = marks;
		this.name = name;
	}
	@Override
	public String toString() {
		return "Student [marks=" + marks + ", name=" + name + "]";
	}

}


public class StudentMain {

	public static void main(String[] args) {
		Student s1=new Student(90,"Hari");
		Student s2=new Student(50,"krishna");
		Student s3=new Student(60,"Ram");
		Student s4=new Student(70,"Hari");
		Student s5=new Student(80,"Shiva");
		ArrayList<Student> as=new ArrayList<>();
		as.add(s1);
		as.add(s2);
		as.add(s3);
		as.add(s4);
		as.add(s5);
		
		Function<Student,Character> f=(s)->{
			if(s.marks >=90) {
				return 'A';
			}else if(s.marks>=80 && s.marks<90) {
				return 'B';
			}
			else if(s.marks>=70 && s.marks<80) {
				return 'C';
			}
			else if(s.marks>=60 && s.marks<70) {
				return 'D';
			}
			else if(s.marks>=50 && s.marks<60) {
				return 'E';
			}else {
				return 'F';
			}
		};
		for(Student s:as) {
			System.out.print(s+" ");
			System.out.print(f.apply(s));
			System.out.println();
		}
		
	}

}

public class FunctionChaining {

	public static void main(String[] args) {
		Function<Integer,Integer> f1=i->i*2;
		Function<Integer,Integer> f2=i->i*i*i;
		
		System.out.println(f1.andThen(f2).apply(2));  //64
		System.out.println(f1.compose(f2).apply(2));  //16
	}

}


f1.andThen(f2) --- first f1 function call and then output f1 as input of  f2 happens
f1.compose(f2) --- first f2 function call and then  output f2 as input of  f1 happens

public class Text {
	

	public static void main(String[] args) {
		Runnable r=()->{
			for(int i=0;i<10;i++) {
				System.out.println("Child thread -1");
			}
		};
		Thread t=new Thread(r);
		t.run();
		for(int i=0;i<10;i++) {
			System.out.println("Parent thread -2");
		}

	}
}

public class Text {
	
	public static void m1() {
		for(int i=0;i<10;i++) {
			System.out.println("Child thread -1");
		}
	}

	public static void main(String[] args) {
		Runnable r=Text::m1;
		Thread t=new Thread(r);
		t.run();
		for(int i=0;i<10;i++) {
			System.out.println("Parent thread -2");
		}

	}

}
