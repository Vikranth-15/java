public class Array {
	public static void main(String[] args) {
		ArrayList<Integer> lis=new ArrayList<>();
		lis.add(11);
		lis.add(21);
		lis.add(1);
		lis.add(4);
		lis.add(8);
		lis.add(18);
		lis.add(10);
		System.out.println(lis);       //[11, 21, 1, 4, 8, 18, 10]

		List<Integer> even=lis.stream()
							  .filter(i-> i%2==0)
							  .collect(Collectors.toList());
		
		System.out.println(even);      // [4, 8, 18, 10]
		
	}
}

public class Array {
	public static void main(String[] args) {
		ArrayList<Integer> lis=new ArrayList<>();
		lis.add(11);
		lis.add(21);
		lis.add(1);
		lis.add(4);
		lis.add(8);
		lis.add(18);
		lis.add(10);
		System.out.println(lis);          // [11, 21, 1, 4, 8, 18, 10]
		List<Integer> res =lis.stream()   
							  .sorted((i1,i2)-> (i1<i2)?1:(i1>i2)?-1:0)
							  .collect(Collectors.toList());
		
		
        System.out.println(res);         // [21, 18, 11, 10, 8, 4, 1] 
		
	}
}

public class MinMax {
	public static void main(String[] args) {
		ArrayList<Integer> lis=new ArrayList<>();
		lis.add(22);
		lis.add(-1);
		lis.add(17);
		lis.add(44);
		lis.add(99);
		lis.add(3);
		lis.add(8);
		lis.add(2);
		lis.add(-19);
		
		Integer min=lis.stream().min((i1,i2)->i1.compareTo(i2)).get();
		System.out.println(min);
		Integer max=lis.stream().max((i1,i2)->i1.compareTo(i2)).get();
		System.out.println(max);
	}

}


.min((i1,i2)-> - i1.compareTo(i2)).get();  gives max 

.max((i1,i2)-> - i1.compareTo(i2)).get();   gives min

public class SmallestNumber {

	public static void main(String[] args) {
		String input = "45 32 67 -89 12 90 -5 100";
		Optional<Integer> min=Arrays.stream(input.split(" "))
			  .map(i->Integer.parseInt(i))
			  .min(Integer::compareTo);
		System.out.println(min.get());
		
		
		// Prints the value if present, avoiding NoSuchElementException on empty strings
        min.ifPresent(System.out::println); 
	}

}

public class MaxWordLength {

	public static void main(String[] args) {
		String sentence = "Coding with Streams is Excellent and super fun";
		Arrays.stream(sentence.split("\\s+"))
        .max(Comparator.comparingInt(String::length))
        .ifPresent(word -> System.out.println("Longest word: " + word + " (" + word.length() + " characters)"));

	}

}

public class SumOfDigits {

	public static void main(String[] args) {
		String s = "1a2b3c4d5e";

		long sum = s.chars()
				.filter(Character::isDigit)
				.map(Character::getNumericValue)
				.reduce(0, (a, b) -> a + b);
		System.out.println(sum);

	}

}
